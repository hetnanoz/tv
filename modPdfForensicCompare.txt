' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    ---
' Returns:       ---
' Description:   Lets the user select one PDF and creates a forensic geometry report.
'-------------------------------------------------------------------------------
Public Sub RunPdfForensicAnalysisBrowse()
    Const METHOD_NAME As String = "RunPdfForensicAnalysisBrowse"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim strOutputPath As String
    Dim strPdfPath As String

    On Error GoTo ErrorHandler

    strPdfPath = SelectPdfFile("Select the PDF to analyse")
    If Len(strPdfPath) = 0 Then
        GoTo ExitSub
    End If

    strOutputPath = BuildForensicOutputPath(strPdfPath)
    If Not PdfForensicAnalyse(strPdfPath, strOutputPath) Then
        Err.Raise ERR_BASE + 1, METHOD_NAME, "The forensic PDF analysis could not be completed."
    End If

    MsgBox _
        "Forensic PDF analysis completed:" & vbCrLf & strOutputPath, _
        vbInformation, _
        "PDF forensic analysis"

ExitSub:
    Exit Sub

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "pdfPath;outputPath", _
        strPdfPath, _
        strOutputPath

    errorManager.save
    errorManager.display "The forensic PDF analysis failed."
    Resume ExitSub
End Sub

' Creation date: 2026-08-06
' Parameters:    pdfPath As String, outputPath As String
' Returns:       Boolean
' Description:   Analyses one PDF and writes detailed vector and text diagnostics.
'-------------------------------------------------------------------------------
Public Function PdfForensicAnalyse( _
    ByVal pdfPath As String, _
    ByVal outputPath As String) As Boolean

    Const METHOD_NAME As String = "PdfForensicAnalyse"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim pdfDoc As tPdfDocument

    On Error GoTo ErrorHandler

    If Len(Trim$(pdfPath)) = 0 Then
        Err.Raise 1001, METHOD_NAME, "The PDF path is required."
    End If

    If Len(Trim$(outputPath)) = 0 Then
        Err.Raise 1020, METHOD_NAME, "The output path is required."
    End If

    Call LoadPdfDocument(pdfPath, pdfDoc)
    If pdfDoc.PageCount = 0 Then
        Err.Raise 1014, METHOD_NAME, "The PDF could not be parsed into page content."
    End If
    Call WriteForensicReport(pdfDoc, outputPath)
    PdfForensicAnalyse = True

ExitFunction:
    Call ResetParserState
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    PdfForensicAnalyse = False

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "pdfPath;outputPath", _
        pdfPath, _
        outputPath

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    oldPdfPath As String, newPdfPath As String, outputPath As String
' Returns:       Boolean
' Description:   Compares two PDFs and writes structural and targeted separator differences.
'-------------------------------------------------------------------------------
Public Function PdfCompare( _
    ByVal oldPdfPath As String, _
    ByVal newPdfPath As String, _
    ByVal outputPath As String) As Boolean

    Const METHOD_NAME As String = "PdfCompare"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim newDoc As tPdfDocument
    Dim oldDoc As tPdfDocument

    On Error GoTo ErrorHandler

    If Len(Trim$(oldPdfPath)) = 0 Then
        Err.Raise 1001, METHOD_NAME, "The old PDF path is required."
    End If

    If Len(Trim$(newPdfPath)) = 0 Then
        Err.Raise 1001, METHOD_NAME, "The new PDF path is required."
    End If

    If Len(Trim$(outputPath)) = 0 Then
        Err.Raise 1020, METHOD_NAME, "The output path is required."
    End If

    Call LoadPdfDocument(oldPdfPath, oldDoc)
    If oldDoc.PageCount = 0 Then
        Err.Raise 1014, METHOD_NAME, "The old PDF could not be parsed into page content."
    End If
    Call LoadPdfDocument(newPdfPath, newDoc)
    If newDoc.PageCount = 0 Then
        Err.Raise 1014, METHOD_NAME, "The new PDF could not be parsed into page content."
    End If
    Call WriteComparisonReport(oldDoc, newDoc, outputPath)
    PdfCompare = True

ExitFunction:
    Call ResetParserState
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    PdfCompare = False

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "oldPdfPath;newPdfPath;outputPath", _
        oldPdfPath, _
        newPdfPath, _
        outputPath

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    pdfPath As String, pdfDoc As tPdfDocument
' Returns:       ---
' Description:   Loads and parses a PDF into a normalized vector and text model.
'-------------------------------------------------------------------------------
Private Sub LoadPdfDocument( _
    ByVal pdfPath As String, _
    ByRef pdfDoc As tPdfDocument)

    Const METHOD_NAME As String = "LoadPdfDocument"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim arrPageRefs() As Long
    Dim lngIndex As Long
    Dim lngPageCount As Long

    On Error GoTo ErrorHandler

    If Len(Dir$(pdfPath)) = 0 Then
        Err.Raise 1010, METHOD_NAME, "The PDF file was not found: " & pdfPath
    End If

    Call ResetParserState
    m_pdf = ReadBytes(pdfPath)
    Call BuildObjectIndex

    pdfDoc.PdfPath = pdfPath
    pdfDoc.FileSize = UBound(m_pdf) + 1
    pdfDoc.PdfVersion = DetectPdfVersion
    pdfDoc.ObjectCount = m_objCount
    pdfDoc.XrefKind = DetectXrefKind
    ReDim pdfDoc.Pages(1 To 8)
    ReDim pdfDoc.Lines(0 To 2047)
    ReDim pdfDoc.Texts(0 To 2047)

    lngPageCount = PageObjectOrder(arrPageRefs)
    If lngPageCount = 0 Then
        Err.Raise 1014, METHOD_NAME, "No PDF page objects were found."
    End If

    For lngIndex = 1 To lngPageCount
        Call ParsePage(arrPageRefs(lngIndex), lngIndex, pdfDoc)
    Next lngIndex

    If pdfDoc.LineCount = 0 And pdfDoc.TextCount = 0 Then
        Err.Raise 1014, METHOD_NAME, "No vector lines or text runs were extracted from the PDF page streams."
    End If

ExitSub:
    Exit Sub

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    pdfDoc.PageCount = 0

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "pdfPath", _
        pdfPath

    errorManager.save
    Resume ExitSub
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    filePath As String
' Returns:       Byte array
' Description:   Reads the complete PDF file into a byte array.
'-------------------------------------------------------------------------------
Private Function ReadBytes(ByVal filePath As String) As Byte()
    Const METHOD_NAME As String = "ReadBytes"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim arrBuffer() As Byte
    Dim blnFileOpen As Boolean
    Dim intFileNumber As Integer
    Dim lngFileSize As Long

    On Error GoTo ErrorHandler

    intFileNumber = FreeFile
    Open filePath For Binary Access Read As #intFileNumber
    blnFileOpen = True
    lngFileSize = LOF(intFileNumber)
    If lngFileSize <= 0 Then
        Err.Raise 1011, METHOD_NAME, "The PDF file is empty."
    End If

    ReDim arrBuffer(0 To lngFileSize - 1)
    Get #intFileNumber, 1, arrBuffer
    ReadBytes = arrBuffer

ExitFunction:
    If blnFileOpen Then
        Close #intFileNumber
    End If
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "filePath", _
        filePath

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    ---
' Returns:       ---
' Description:   Builds an index of indirect PDF objects.
'-------------------------------------------------------------------------------
Private Sub BuildObjectIndex()
    Const METHOD_NAME As String = "BuildObjectIndex"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim arrParts() As String
    Dim blnGenerationFound As Boolean
    Dim lngIndex As Long
    Dim lngPosition As Long
    Dim strHeader As String
    Dim strObjectNumber As String

    On Error GoTo ErrorHandler

    m_objCount = 0
    ReDim m_objNum(0 To 255)
    ReDim m_objPos(0 To 255)
    lngPosition = 0

    Do
        lngPosition = FindBytes(m_pdf, lngPosition, " obj")
        If lngPosition < 0 Then
            Exit Do
        End If

        strHeader = BytesText(m_pdf, IIf(lngPosition - 24 < 0, 0, lngPosition - 24), lngPosition - 1)
        strHeader = Replace(Replace(strHeader, vbCr, " "), vbLf, " ")
        arrParts = Split(Trim$(strHeader), " ")
        strObjectNumber = vbNullString
        blnGenerationFound = False

        For lngIndex = UBound(arrParts) To LBound(arrParts) Step -1
            If Len(arrParts(lngIndex)) > 0 Then
                If IsNumeric(arrParts(lngIndex)) Then
                    If Not blnGenerationFound Then
                        blnGenerationFound = True
                    Else
                        strObjectNumber = arrParts(lngIndex)
                        Exit For
                    End If
                Else
                    Exit For
                End If
            End If
        Next lngIndex

        If Len(strObjectNumber) > 0 Then
            If m_objCount > UBound(m_objNum) Then
                ReDim Preserve m_objNum(0 To m_objCount * 2)
                ReDim Preserve m_objPos(0 To m_objCount * 2)
            End If
            m_objNum(m_objCount) = CLng(strObjectNumber)
            m_objPos(m_objCount) = lngPosition + 4
            m_objCount = m_objCount + 1
        End If

        lngPosition = lngPosition + 4
    Loop

ExitSub:
    Exit Sub

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription

    errorManager.save
    Resume ExitSub
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    objectPosition As Long
' Returns:       String
' Description:   Returns the dictionary text of an indirect PDF object.
'-------------------------------------------------------------------------------
Private Function ObjectDict(ByVal objectPosition As Long) As String
    Const METHOD_NAME As String = "ObjectDict"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngEndObject As Long
    Dim lngEndStream As Long
    Dim lngStopPosition As Long

    On Error GoTo ErrorHandler

    lngEndStream = FindBytes(m_pdf, objectPosition, "stream")
    lngEndObject = FindBytes(m_pdf, objectPosition, "endobj")

    If lngEndStream < 0 Then
        lngStopPosition = lngEndObject
    ElseIf lngEndObject < 0 Then
        lngStopPosition = lngEndStream
    ElseIf lngEndStream < lngEndObject Then
        lngStopPosition = lngEndStream
    Else
        lngStopPosition = lngEndObject
    End If

    If lngStopPosition < 0 Then
        lngStopPosition = objectPosition + 4000
    End If
    If lngStopPosition - objectPosition > 8000 Then
        lngStopPosition = objectPosition + 8000
    End If

    ObjectDict = Replace(Replace(BytesText(m_pdf, objectPosition, lngStopPosition - 1), vbCr, " "), vbLf, " ")

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    ObjectDict = vbNullString

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "objectPosition", _
        objectPosition

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    objectPosition As Long, dataStart As Long, dataEnd As Long
' Returns:       Boolean
' Description:   Locates the byte bounds of a PDF stream.
'-------------------------------------------------------------------------------
Private Function StreamBounds( _
    ByVal objectPosition As Long, _
    ByRef dataStart As Long, _
    ByRef dataEnd As Long) As Boolean

    Const METHOD_NAME As String = "StreamBounds"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngDeclaredLength As Long
    Dim lngEndObject As Long
    Dim lngEndStream As Long
    Dim lngStreamPosition As Long
    Dim strProbe As String

    On Error GoTo ErrorHandler

    lngStreamPosition = FindBytes(m_pdf, objectPosition, "stream")
    lngEndObject = FindBytes(m_pdf, objectPosition, "endobj")
    If lngStreamPosition < 0 Then
        GoTo ExitFunction
    End If
    If lngEndObject >= 0 And lngStreamPosition > lngEndObject Then
        GoTo ExitFunction
    End If

    dataStart = lngStreamPosition + 6
    If dataStart > UBound(m_pdf) Then
        GoTo ExitFunction
    End If
    If m_pdf(dataStart) = 13 Then
        dataStart = dataStart + 1
    End If
    If m_pdf(dataStart) = 10 Then
        dataStart = dataStart + 1
    End If

    lngDeclaredLength = DeclaredLength(ObjectDict(objectPosition))
    If lngDeclaredLength > 0 Then
        If dataStart + lngDeclaredLength + 20 <= UBound(m_pdf) Then
            strProbe = BytesText(m_pdf, dataStart + lngDeclaredLength, dataStart + lngDeclaredLength + 20)
            If InStr(1, strProbe, "endstream", vbBinaryCompare) > 0 Then
                dataEnd = dataStart + lngDeclaredLength - 1
                StreamBounds = True
                GoTo ExitFunction
            End If
        End If
    End If

    lngEndStream = FindBytes(m_pdf, dataStart, "endstream")
    If lngEndStream < 0 Then
        GoTo ExitFunction
    End If

    dataEnd = lngEndStream - 1
    Do While dataEnd >= dataStart
        If m_pdf(dataEnd) <> 10 And m_pdf(dataEnd) <> 13 Then
            Exit Do
        End If
        dataEnd = dataEnd - 1
    Loop

    StreamBounds = True

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    StreamBounds = False

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "objectPosition", _
        objectPosition

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    dictionaryText As String
' Returns:       Long
' Description:   Returns a direct numeric stream length from a PDF dictionary.
'-------------------------------------------------------------------------------
Private Function DeclaredLength(ByVal dictionaryText As String) As Long
    Const METHOD_NAME As String = "DeclaredLength"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngIndex As Long
    Dim lngKeyPosition As Long
    Dim strCharacter As String
    Dim strNumber As String
    Dim strTail As String

    On Error GoTo ErrorHandler

    lngKeyPosition = InStr(1, dictionaryText, "/Length", vbBinaryCompare)
    If lngKeyPosition = 0 Then
        GoTo ExitFunction
    End If

    lngIndex = lngKeyPosition + 7
    Do While lngIndex <= Len(dictionaryText)
        If Mid$(dictionaryText, lngIndex, 1) <> " " Then
            Exit Do
        End If
        lngIndex = lngIndex + 1
    Loop

    Do While lngIndex <= Len(dictionaryText)
        strCharacter = Mid$(dictionaryText, lngIndex, 1)
        If strCharacter < "0" Or strCharacter > "9" Then
            Exit Do
        End If
        strNumber = strNumber & strCharacter
        lngIndex = lngIndex + 1
    Loop

    If Len(strNumber) = 0 Then
        GoTo ExitFunction
    End If

    strTail = LTrim$(Mid$(dictionaryText, lngIndex, 8))
    If InStr(1, strTail, " R", vbBinaryCompare) > 0 Then
        GoTo ExitFunction
    End If

    DeclaredLength = CLng(strNumber)

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    DeclaredLength = 0

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    objectPosition As Long
' Returns:       String
' Description:   Returns an uncompressed text representation of a PDF stream.
'-------------------------------------------------------------------------------
Private Function StreamText(ByVal objectPosition As Long) As String
    Const METHOD_NAME As String = "StreamText"
    Dim errorManager As New ErrorManager
    Dim arrDecoded() As Byte
    Dim arrWorking() As Byte
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngDataEnd As Long
    Dim lngDataStart As Long
    Dim lngDecodedLength As Long
    Dim lngWorkingLength As Long
    Dim strDecodeError As String
    Dim strDictionary As String

    On Error GoTo ErrorHandler

    If Not StreamBounds(objectPosition, lngDataStart, lngDataEnd) Then
        GoTo ExitFunction
    End If

    strDictionary = ObjectDict(objectPosition)
    Call CopyByteRange(m_pdf, lngDataStart, lngDataEnd, arrWorking, lngWorkingLength)

    If InStr(1, strDictionary, "/ASCII85Decode", vbBinaryCompare) > 0 Or _
       InStr(1, strDictionary, "/A85", vbBinaryCompare) > 0 Then

        If Not DecodeAscii85Bytes(arrWorking, lngWorkingLength, arrDecoded, lngDecodedLength) Then
            Err.Raise 1027, METHOD_NAME, "The PDF ASCII85 stream could not be decoded."
        End If
        Call ReplaceWorkingBytes(arrWorking, lngWorkingLength, arrDecoded, lngDecodedLength)
    End If

    If InStr(1, strDictionary, "/ASCIIHexDecode", vbBinaryCompare) > 0 Or _
       InStr(1, strDictionary, "/AHx", vbBinaryCompare) > 0 Then

        If Not DecodeAsciiHexBytes(arrWorking, lngWorkingLength, arrDecoded, lngDecodedLength) Then
            Err.Raise 1027, METHOD_NAME, "The PDF ASCIIHex stream could not be decoded."
        End If
        Call ReplaceWorkingBytes(arrWorking, lngWorkingLength, arrDecoded, lngDecodedLength)
    End If

    If InStr(1, strDictionary, "/FlateDecode", vbBinaryCompare) > 0 Or _
       InStr(1, strDictionary, "/Fl", vbBinaryCompare) > 0 Then

        If Not TryInflateBytes( _
            arrWorking, _
            lngWorkingLength, _
            arrDecoded, _
            lngDecodedLength, _
            strDecodeError) Then

            Err.Raise _
                1027, _
                METHOD_NAME, _
                "The PDF Flate stream could not be decoded. " & strDecodeError
        End If
        Call ReplaceWorkingBytes(arrWorking, lngWorkingLength, arrDecoded, lngDecodedLength)
    End If

    If InStr(1, strDictionary, "/LZWDecode", vbBinaryCompare) > 0 Then
        Err.Raise 1027, METHOD_NAME, "LZWDecode is not supported by the forensic comparison module."
    End If

    StreamText = ByteArrayText(arrWorking, lngWorkingLength)

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    StreamText = vbNullString

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "objectPosition;filterDictionary", _
        objectPosition, _
        Left$(strDictionary, 500)

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    sourceBytes, sourceLength, resultBytes, resultLength, decodeError
' Returns:       Boolean
' Description:   Inflates either zlib-wrapped or raw DEFLATE bytes without logging expected retries.
'-------------------------------------------------------------------------------
Private Function TryInflateBytes( _
    ByRef sourceBytes() As Byte, _
    ByVal sourceLength As Long, _
    ByRef resultBytes() As Byte, _
    ByRef resultLength As Long, _
    ByRef decodeError As String) As Boolean

    Const METHOD_NAME As String = "TryInflateBytes"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngAttemptError As Long
    Dim lngSkip As Long
    Dim strAttemptError As String

    On Error GoTo ErrorHandler

    resultLength = 0
    decodeError = vbNullString
    If sourceLength <= 0 Then
        decodeError = "The stream is empty."
        GoTo ExitFunction
    End If

    lngSkip = ZlibSkipBytes(sourceBytes, sourceLength)

    On Error Resume Next
    VBA.Err.Clear
    resultLength = modPdfReport.InflateStream( _
        sourceBytes, _
        lngSkip, _
        sourceLength - 1, _
        resultBytes)
    lngAttemptError = VBA.Err.Number
    strAttemptError = VBA.Err.Description
    VBA.Err.Clear
    On Error GoTo ErrorHandler

    If lngAttemptError = 0 And resultLength > 0 Then
        TryInflateBytes = True
        GoTo ExitFunction
    End If

    If lngSkip <> 0 Then
        On Error Resume Next
        VBA.Err.Clear
        resultLength = modPdfReport.InflateStream( _
            sourceBytes, _
            0, _
            sourceLength - 1, _
            resultBytes)
        lngAttemptError = VBA.Err.Number
        strAttemptError = VBA.Err.Description
        VBA.Err.Clear
        On Error GoTo ErrorHandler

        If lngAttemptError = 0 And resultLength > 0 Then
            TryInflateBytes = True
            GoTo ExitFunction
        End If
    End If

    decodeError = "DEFLATE decoder error " & CStr(lngAttemptError) & ": " & strAttemptError

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    resultLength = 0
    TryInflateBytes = False
    decodeError = errDescription

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "sourceLength;zlibSkip", _
        sourceLength, _
        lngSkip

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    sourceBytes, sourceLength, resultBytes, resultLength
' Returns:       Boolean
' Description:   Decodes a PDF ASCII85 stream, including z shorthand and partial groups.
'-------------------------------------------------------------------------------
Private Function DecodeAscii85Bytes( _
    ByRef sourceBytes() As Byte, _
    ByVal sourceLength As Long, _
    ByRef resultBytes() As Byte, _
    ByRef resultLength As Long) As Boolean

    Const METHOD_NAME As String = "DecodeAscii85Bytes"
    Dim errorManager As New ErrorManager
    Dim arrGroup(0 To 4) As Long
    Dim dblValue As Double
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngByte As Long
    Dim lngGroupCount As Long
    Dim lngIndex As Long
    Dim lngOriginalCount As Long
    Dim lngOutputIndex As Long

    On Error GoTo ErrorHandler

    resultLength = 0
    ReDim resultBytes(0 To sourceLength + 3)

    For lngIndex = 0 To sourceLength - 1
        lngByte = CLng(sourceBytes(lngIndex))

        If IsPdfWhiteSpaceByte(lngByte) Then
            GoTo ContinueLoop
        End If

        If lngByte = 60 Then
            If lngIndex + 1 < sourceLength Then
                If sourceBytes(lngIndex + 1) = 126 Then
                    lngIndex = lngIndex + 1
                    GoTo ContinueLoop
                End If
            End If
        End If

        If lngByte = 126 Then
            Exit For
        End If

        If lngByte = 122 Then
            If lngGroupCount <> 0 Then
                Err.Raise 1027, METHOD_NAME, "ASCII85 z shorthand appeared inside a partial group."
            End If
            resultBytes(lngOutputIndex) = 0
            resultBytes(lngOutputIndex + 1) = 0
            resultBytes(lngOutputIndex + 2) = 0
            resultBytes(lngOutputIndex + 3) = 0
            lngOutputIndex = lngOutputIndex + 4
            GoTo ContinueLoop
        End If

        If lngByte < 33 Or lngByte > 117 Then
            Err.Raise 1027, METHOD_NAME, "Invalid ASCII85 character code: " & CStr(lngByte)
        End If

        arrGroup(lngGroupCount) = lngByte - 33
        lngGroupCount = lngGroupCount + 1

        If lngGroupCount = 5 Then
            dblValue = Ascii85GroupValue(arrGroup)
            Call AppendAscii85Value(resultBytes, lngOutputIndex, dblValue, 4)
            lngGroupCount = 0
        End If
ContinueLoop:
    Next lngIndex

    If lngGroupCount = 1 Then
        Err.Raise 1027, METHOD_NAME, "ASCII85 final group contains only one character."
    End If

    If lngGroupCount > 1 Then
        lngOriginalCount = lngGroupCount
        Do While lngGroupCount < 5
            arrGroup(lngGroupCount) = 84
            lngGroupCount = lngGroupCount + 1
        Loop
        dblValue = Ascii85GroupValue(arrGroup)
        Call AppendAscii85Value(resultBytes, lngOutputIndex, dblValue, lngOriginalCount - 1)
    End If

    resultLength = lngOutputIndex
    If resultLength = 0 Then
        ReDim resultBytes(0 To 0)
    Else
        ReDim Preserve resultBytes(0 To resultLength - 1)
    End If
    DecodeAscii85Bytes = True

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    resultLength = 0
    ReDim resultBytes(0 To 0)
    DecodeAscii85Bytes = False

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "sourceLength;inputIndex", _
        sourceLength, _
        lngIndex

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    groupValues
' Returns:       Double
' Description:   Converts five ASCII85 digits into one unsigned 32-bit value.
'-------------------------------------------------------------------------------
Private Function Ascii85GroupValue(ByRef groupValues() As Long) As Double
    Const METHOD_NAME As String = "Ascii85GroupValue"

    Dim errDescription As String

        Err.Raise 1027, METHOD_NAME, "ASCII85 group exceeds the unsigned 32-bit range."
    End If
    Ascii85GroupValue = dblValue

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Ascii85GroupValue = 0#

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    resultBytes, outputIndex, unsignedValue, byteCount
' Returns:       ---
' Description:   Appends up to four big-endian bytes from an unsigned 32-bit value.
'-------------------------------------------------------------------------------
Private Sub AppendAscii85Value( _
    ByRef resultBytes() As Byte, _
    ByRef outputIndex As Long, _
    ByVal unsignedValue As Double, _
    ByVal byteCount As Long)

    Const METHOD_NAME As String = "AppendAscii85Value"
    Dim errorManager As New ErrorManager
    Dim arrValues(0 To 3) As Long
    Dim dblRemainder As Double
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngIndex As Long

    On Error GoTo ErrorHandler

    dblRemainder = unsignedValue
    arrValues(0) = Fix(dblRemainder / 16777216#)
    dblRemainder = dblRemainder - CDbl(arrValues(0)) * 16777216#
    arrValues(1) = Fix(dblRemainder / 65536#)
    dblRemainder = dblRemainder - CDbl(arrValues(1)) * 65536#
    arrValues(2) = Fix(dblRemainder / 256#)
    dblRemainder = dblRemainder - CDbl(arrValues(2)) * 256#
    arrValues(3) = Fix(dblRemainder)

    For lngIndex = 0 To byteCount - 1
        resultBytes(outputIndex) = CByte(arrValues(lngIndex))
        outputIndex = outputIndex + 1
    Next lngIndex

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    sourceBytes, sourceLength, resultBytes, resultLength
' Returns:       Boolean
' Description:   Decodes a PDF ASCIIHex stream.
'-------------------------------------------------------------------------------
Private Function DecodeAsciiHexBytes( _
    ByRef sourceBytes() As Byte, _
    ByVal sourceLength As Long, _
    ByRef resultBytes() As Byte, _
    ByRef resultLength As Long) As Boolean

    Const METHOD_NAME As String = "DecodeAsciiHexBytes"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim intHighNibble As Integer
    Dim intValue As Integer
    Dim lngByte As Long
    Dim lngIndex As Long
    Dim lngOutputIndex As Long

    On Error GoTo ErrorHandler

    intHighNibble = -1
    ReDim resultBytes(0 To sourceLength \ 2 + 1)

    For lngIndex = 0 To sourceLength - 1
        lngByte = CLng(sourceBytes(lngIndex))
        If IsPdfWhiteSpaceByte(lngByte) Then
            GoTo ContinueLoop
        End If
        If lngByte = 62 Then
            Exit For
        End If

        intValue = HexByteValue(lngByte)
        If intValue < 0 Then
            Err.Raise 1027, METHOD_NAME, "Invalid ASCIIHex character code: " & CStr(lngByte)
        End If

        If intHighNibble < 0 Then
            intHighNibble = intValue
        Else
            resultBytes(lngOutputIndex) = CByte(intHighNibble * 16 + intValue)
            lngOutputIndex = lngOutputIndex + 1
            intHighNibble = -1
        End If
ContinueLoop:
    Next lngIndex

    If intHighNibble >= 0 Then
        resultBytes(lngOutputIndex) = CByte(intHighNibble * 16)
        lngOutputIndex = lngOutputIndex + 1
    End If

    resultLength = lngOutputIndex
    If resultLength = 0 Then
        ReDim resultBytes(0 To 0)
    Else
        ReDim Preserve resultBytes(0 To resultLength - 1)
    End If
    DecodeAsciiHexBytes = True

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    resultLength = 0
    ReDim resultBytes(0 To 0)
    DecodeAsciiHexBytes = False

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "sourceLength;inputIndex", _
        sourceLength, _
        lngIndex

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    dictionaryText As String, keyName As String
' Returns:       String
' Description:   Returns all indirect object references following a PDF dictionary key.
'-------------------------------------------------------------------------------
Private Function RefList( _
    ByVal dictionaryText As String, _
    ByVal keyName As String) As String

    Const METHOD_NAME As String = "RefList"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim arrParts() As String
    Dim lngEndPosition As Long
    Dim lngIndex As Long
    Dim lngKeyPosition As Long
    Dim lngStartPosition As Long
    Dim strResult As String
    Dim strSegment As String

    On Error GoTo ErrorHandler

    lngKeyPosition = InStr(1, dictionaryText, keyName, vbBinaryCompare)
    If lngKeyPosition = 0 Then
        GoTo ExitFunction
    End If

    lngStartPosition = lngKeyPosition + Len(keyName)
    Do While lngStartPosition <= Len(dictionaryText)
        If Mid$(dictionaryText, lngStartPosition, 1) <> " " Then
            Exit Do
        End If
        lngStartPosition = lngStartPosition + 1
    Loop

    If Mid$(dictionaryText, lngStartPosition, 1) = "[" Then
        lngEndPosition = InStr(lngStartPosition, dictionaryText, "]", vbBinaryCompare)
        If lngEndPosition = 0 Then
            lngEndPosition = lngStartPosition + 1000
        End If
        strSegment = Mid$(dictionaryText, lngStartPosition + 1, lngEndPosition - lngStartPosition - 1)
    Else
        strSegment = Mid$(dictionaryText, lngStartPosition, 80)
        lngEndPosition = InStr(1, strSegment, "/", vbBinaryCompare)
        If lngEndPosition > 0 Then
            strSegment = Left$(strSegment, lngEndPosition - 1)
        End If
    End If

    Do While InStr(1, strSegment, "  ", vbBinaryCompare) > 0
        strSegment = Replace(strSegment, "  ", " ")
    Loop

    arrParts = Split(Trim$(strSegment), " ")
    For lngIndex = LBound(arrParts) To UBound(arrParts)
        If arrParts(lngIndex) = "R" And lngIndex >= 2 Then
            If Len(strResult) > 0 Then
                strResult = strResult & ","
            End If
            strResult = strResult & arrParts(lngIndex - 2)
        End If
    Next lngIndex

    RefList = strResult

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    RefList = vbNullString

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "keyName", _
        keyName

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    dictionaryText As String, keyName As String
' Returns:       String
' Description:   Returns the compact value following a PDF dictionary key.
'-------------------------------------------------------------------------------
Private Function DictValue( _
    ByVal dictionaryText As String, _
    ByVal keyName As String) As String

    Const METHOD_NAME As String = "DictValue"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngDepth As Long
    Dim lngIndex As Long
    Dim lngKeyPosition As Long
    Dim strCharacter As String
    Dim strValue As String

    On Error GoTo ErrorHandler

    lngKeyPosition = InStr(1, dictionaryText, keyName, vbBinaryCompare)
    If lngKeyPosition = 0 Then
        DictValue = "(absent)"
        GoTo ExitFunction
    End If

    lngIndex = lngKeyPosition + Len(keyName)
    Do While lngIndex <= Len(dictionaryText)
        If Mid$(dictionaryText, lngIndex, 1) <> " " Then
            Exit Do
        End If
        lngIndex = lngIndex + 1
    Loop

    Do While lngIndex <= Len(dictionaryText)
        strCharacter = Mid$(dictionaryText, lngIndex, 1)
        strValue = strValue & strCharacter
        If strCharacter = "[" Or strCharacter = "<" Then
            lngDepth = lngDepth + 1
        End If
        If strCharacter = "]" Or strCharacter = ">" Then
            lngDepth = lngDepth - 1
            If lngDepth <= 0 Then
                Exit Do
            End If
        End If
        If lngDepth = 0 And Len(strValue) > 1 Then
            If strCharacter = " " Or strCharacter = "/" Then
                Exit Do
            End If
        End If
        If Len(strValue) > 1000 Then
            Exit Do
        End If
        lngIndex = lngIndex + 1
    Loop

    DictValue = Trim$(strValue)

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    DictValue = "(error)"

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "keyName", _
        keyName

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    objectNo As Long, pageNo As Long, pdfDoc As tPdfDocument
' Returns:       ---
' Description:   Parses one page content stream into normalized line and text records.
'-------------------------------------------------------------------------------
Private Sub ParsePage( _
    ByVal objectNo As Long, _
    ByVal pageNo As Long, _
    ByRef pdfDoc As tPdfDocument)

    Const METHOD_NAME As String = "ParsePage"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngObjectIndex As Long
    Dim strContent As String
    Dim strDictionary As String

    On Error GoTo ErrorHandler

    lngObjectIndex = IndexOfObject(objectNo)
    If lngObjectIndex < 0 Then
        Err.Raise 1014, METHOD_NAME, "The page object was not found."
    End If

    strDictionary = ObjectDict(m_objPos(lngObjectIndex))
    strContent = PageContent(strDictionary)

    pdfDoc.PageCount = pdfDoc.PageCount + 1
    If pdfDoc.PageCount > UBound(pdfDoc.Pages) Then
        ReDim Preserve pdfDoc.Pages(1 To UBound(pdfDoc.Pages) * 2)
    End If

    With pdfDoc.Pages(pdfDoc.PageCount)
        .ObjectNo = objectNo
        .MediaBox = DictValue(strDictionary, "/MediaBox")
        .CropBox = DictValue(strDictionary, "/CropBox")
        .RotateValue = DictValue(strDictionary, "/Rotate")
        .FontDictionary = DictValue(strDictionary, "/Font")
        .ContentLength = Len(strContent)
        .LineStart = pdfDoc.LineCount
        .TextStart = pdfDoc.TextCount
    End With

    Call AnalysePageContent(strContent, pageNo, pdfDoc)

    pdfDoc.Pages(pdfDoc.PageCount).LineCount = pdfDoc.LineCount - pdfDoc.Pages(pdfDoc.PageCount).LineStart
    pdfDoc.Pages(pdfDoc.PageCount).TextCount = pdfDoc.TextCount - pdfDoc.Pages(pdfDoc.PageCount).TextStart

ExitSub:
    Exit Sub

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "objectNo;pageNo", _
        objectNo, _
        pageNo

    errorManager.save
    Resume ExitSub
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    content As String, pageNo As Long, pdfDoc As tPdfDocument
' Returns:       ---
' Description:   Interprets relevant PDF graphics and text operators for one page.
'-------------------------------------------------------------------------------
Private Sub AnalysePageContent( _
    ByVal content As String, _
    ByVal pageNo As Long, _
    ByRef pdfDoc As tPdfDocument)

    Const METHOD_NAME As String = "AnalysePageContent"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim arrTokens() As String
    Dim blnInArray As Boolean
    Dim dblFontSize As Double
    Dim dblLineX As Double
    Dim dblLineY As Double
    Dim dblX1 As Double
    Dim dblX2 As Double
    Dim dblY1 As Double
    Dim dblY2 As Double
    Dim h0 As String
    Dim h1 As String
    Dim h2 As String
    Dim h3 As String
    Dim h4 As String
    Dim h5 As String
    Dim lngIndex As Long
    Dim lngTokenCount As Long
    Dim strArrayText As String
    Dim strFontName As String
    Dim strLastText As String
    Dim strToken As String

    On Error GoTo ErrorHandler

    Call ResetPageGraphicsState
    lngTokenCount = Tokenize(content, arrTokens)

    For lngIndex = 0 To lngTokenCount - 1
        strToken = arrTokens(lngIndex)

        Select Case strToken
            Case "q"
                Call PushGraphicsState

            Case "Q"
                Call PopGraphicsState

            Case "cm"
                Call ApplyConcatMatrix(Val(h5), Val(h4), Val(h3), Val(h2), Val(h1), Val(h0))

            Case "w"
                m_currentLineWidth = Val(h0)

            Case "G"
                m_currentStrokeColor = "G:" & NumTxt(Val(h0))

            Case "RG"
                m_currentStrokeColor = "RGB:" & NumTxt(Val(h2)) & "," & NumTxt(Val(h1)) & "," & NumTxt(Val(h0))

            Case "K"
                m_currentStrokeColor = "CMYK:" & NumTxt(Val(h3)) & "," & NumTxt(Val(h2)) & "," & NumTxt(Val(h1)) & "," & NumTxt(Val(h0))

            Case "m"
                Call TransformPoint(Val(h1), Val(h0), dblX1, dblY1)
                m_moveX = dblX1
                m_moveY = dblY1
                m_subPathStartX = dblX1
                m_subPathStartY = dblY1
                m_hasMovePoint = True

            Case "l"
                Call TransformPoint(Val(h1), Val(h0), dblX2, dblY2)
                If m_hasMovePoint Then
                    Call AppendPendingLine(m_moveX, m_moveY, dblX2, dblY2, "m/l")
                End If
                m_moveX = dblX2
                m_moveY = dblY2

            Case "h"
                If m_hasMovePoint Then
                    Call AppendPendingLine(m_moveX, m_moveY, m_subPathStartX, m_subPathStartY, "h")
                    m_moveX = m_subPathStartX
                    m_moveY = m_subPathStartY
                End If

            Case "re"
                Call AddRectanglePath(Val(h3), Val(h2), Val(h1), Val(h0))

            Case "S"
                Call FlushPendingPath(pageNo, "S", pdfDoc)

            Case "s"
                If m_hasMovePoint Then
                    Call AppendPendingLine(m_moveX, m_moveY, m_subPathStartX, m_subPathStartY, "h")
                End If
                Call FlushPendingPath(pageNo, "s", pdfDoc)

            Case "B", "B*", "b", "b*"
                Call FlushPendingPath(pageNo, strToken, pdfDoc)

            Case "f", "f*", "F", "n"
                Call ClearPendingPath

            Case "BT"
                dblLineX = 0
                dblLineY = 0

            Case "Tf"
                dblFontSize = Val(h0)
                strFontName = h1

            Case "Tm"
                dblLineX = Val(h1)
                dblLineY = Val(h0)

            Case "Td", "TD"
                dblLineX = dblLineX + Val(h1)
                dblLineY = dblLineY + Val(h0)

            Case "T*"
                dblLineY = dblLineY - dblFontSize * 1.2

            Case "Tj", "'"
                Call AppendDocumentText(pageNo, dblLineX, dblLineY, dblFontSize, strFontName, strLastText, pdfDoc)

            Case "TJ"
                Call AppendDocumentText(pageNo, dblLineX, dblLineY, dblFontSize, strFontName, strArrayText, pdfDoc)
                strArrayText = vbNullString

            Case "["
                blnInArray = True
                strArrayText = vbNullString

            Case "]"
                blnInArray = False
        End Select

        If Len(strToken) > 0 Then
            If Left$(strToken, 1) = "(" Then
                strLastText = Mid$(strToken, 2)
                If blnInArray Then
                    strArrayText = strArrayText & strLastText
                End If
            End If
        End If

        h5 = h4
        h4 = h3
        h3 = h2
        h2 = h1
        h1 = h0
        h0 = strToken
    Next lngIndex

    Call ClearPendingPath

ExitSub:
    Exit Sub

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "pageNo;contentLength", _
        pageNo, _
        Len(content)

    errorManager.save
    Resume ExitSub
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    a As Double, b As Double, c As Double, d As Double, e As Double, f As Double
' Returns:       ---
' Description:   Concatenates one PDF transformation matrix with the current matrix.
'-------------------------------------------------------------------------------
Private Sub ApplyConcatMatrix( _
    ByVal a As Double, _
    ByVal b As Double, _
    ByVal c As Double, _
    ByVal d As Double, _
    ByVal e As Double, _
    ByVal f As Double)

    Dim errDescription As String

    Dim dblNewA As Double
    Dim dblNewB As Double
    Dim dblNewC As Double
    Dim dblNewD As Double
    Dim dblNewE As Double
    Dim dblNewF As Double

    On Error GoTo ErrorHandler

    dblNewA = a * m_ctmA + b * m_ctmC
    dblNewB = a * m_ctmB + b * m_ctmD
    dblNewC = c * m_ctmA + d * m_ctmC
    dblNewD = c * m_ctmB + d * m_ctmD
    dblNewE = e * m_ctmA + f * m_ctmC + m_ctmE
    dblNewF = e * m_ctmB + f * m_ctmD + m_ctmF

    m_ctmA = dblNewA
    m_ctmB = dblNewB
    m_ctmC = dblNewC
    m_ctmD = dblNewD
    m_ctmE = dblNewE
    m_ctmF = dblNewF

ExitSub:
    Exit Sub

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription

    errorManager.save
    Resume ExitSub
End Sub

'-------------------------------------------------------------------------------

    errorManager.addError _

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    content As String, tokens As String array
' Returns:       Long
' Description:   Tokenizes PDF page content while preserving literal strings.
'-------------------------------------------------------------------------------
Private Function Tokenize( _
    ByVal content As String, _
    ByRef tokens() As String) As Long

    Const METHOD_NAME As String = "Tokenize"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngDepth As Long
    Dim lngIndex As Long
    Dim lngLength As Long
    Dim lngNext As Long
    Dim strBuffer As String
    Dim strCharacter As String

    On Error GoTo ErrorHandler

    lngLength = Len(content)
    ReDim tokens(0 To 4095)
    lngIndex = 1

    Do While lngIndex <= lngLength
        strCharacter = Mid$(content, lngIndex, 1)

        If strCharacter = " " Or strCharacter = vbCr Or strCharacter = vbLf Or strCharacter = vbTab Or strCharacter = Chr$(0) Then
            lngIndex = lngIndex + 1

        ElseIf strCharacter = "(" Then
            lngDepth = 1
            lngNext = lngIndex + 1
            strBuffer = vbNullString
            Do While lngNext <= lngLength And lngDepth > 0
                strCharacter = Mid$(content, lngNext, 1)
                If strCharacter = "\" Then
                    strBuffer = strBuffer & Mid$(content, lngNext + 1, 1)
                    lngNext = lngNext + 2
                Else
                    If strCharacter = "(" Then
                        lngDepth = lngDepth + 1
                    ElseIf strCharacter = ")" Then
                        lngDepth = lngDepth - 1
                        If lngDepth = 0 Then
                            Exit Do
                        End If
                    End If
                    strBuffer = strBuffer & strCharacter
                    lngNext = lngNext + 1
                End If
            Loop
            Call PushToken(tokens, Tokenize, "(" & strBuffer)
            lngIndex = lngNext + 1

        ElseIf strCharacter = "/" Then
            lngNext = lngIndex + 1
            Do While lngNext <= lngLength
                If IsDelimiter(Mid$(content, lngNext, 1)) Then
                    Exit Do
                End If
                lngNext = lngNext + 1
            Loop
            Call PushToken(tokens, Tokenize, Mid$(content, lngIndex, lngNext - lngIndex))
            lngIndex = lngNext

        ElseIf strCharacter = "[" Or strCharacter = "]" Then
            Call PushToken(tokens, Tokenize, strCharacter)
            lngIndex = lngIndex + 1

        ElseIf strCharacter = "<" Or strCharacter = ">" Or strCharacter = "{" Or strCharacter = "}" Then
            lngIndex = lngIndex + 1

        ElseIf strCharacter = "%" Then
            lngNext = InStr(lngIndex, content, vbLf, vbBinaryCompare)
            If lngNext = 0 Then
                Exit Do
            End If
            lngIndex = lngNext + 1

        Else
            lngNext = lngIndex
            Do While lngNext <= lngLength
                If IsDelimiter(Mid$(content, lngNext, 1)) Then
                    Exit Do
                End If
                lngNext = lngNext + 1
            Loop
            If lngNext = lngIndex Then
                lngNext = lngIndex + 1
            End If
            Call PushToken(tokens, Tokenize, Mid$(content, lngIndex, lngNext - lngIndex))
            lngIndex = lngNext
        End If
    Loop

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    Tokenize = 0

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "contentLength", _
        Len(content)

    errorManager.save
    Resume ExitFunction
End Function

Private Sub WriteDuplicateSection( _

    ByVal pageNo As Long, _

    Const METHOD_NAME As String = "WriteDuplicateSection"
    Dim errorManager As New ErrorManager

    Dim lngDetailCount As Long
    Dim lngDuplicateGroups As Long
    Dim lngIndex As Long
    Dim strKey As String
    Dim varKey As Variant

    On Error GoTo ErrorHandler

    Set dictCounts = CreateObject("Scripting.Dictionary")
    For lngIndex = 0 To pdfDoc.LineCount - 1
        If pdfDoc.Lines(lngIndex).PageNo = pageNo Then
            strKey = GeometryKey(pdfDoc.Lines(lngIndex))
            If dictCounts.Exists(strKey) Then
                dictCounts(strKey) = CLng(dictCounts(strKey)) + 1
            Else
                dictCounts.Add strKey, 1
            End If
        End If
    Next lngIndex

            lngDuplicateGroups = lngDuplicateGroups + 1
        End If
    Next varKey

    Call EmitLine(fileNumber, "DUPLICATE_STROKES|groups=" & CStr(lngDuplicateGroups))
    For Each varKey In dictCounts.Keys
        If CLng(dictCounts(varKey)) > 1 Then
            lngDetailCount = lngDetailCount + 1
            If lngDetailCount <= MAX_DETAIL_LINES Then
                Call EmitLine(fileNumber, "DUPLICATE|count=" & CStr(dictCounts(varKey)) & "|" & CStr(varKey))
            End If
        End If
    Next varKey

ExitSub:
    Set dictCounts = Nothing
    Exit Sub

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description

    errorManager.addError _
        CLASS_NAME, _

        errNumber, _
        errDescription, _

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    pdfDoc As tPdfDocument, pageNo As Long, fileNumber As Integer
' Returns:       ---
' Description:   Writes all estimated intersections between vertical rules and text runs.
'-------------------------------------------------------------------------------
Private Sub WriteCrossingSection( _
    ByRef pdfDoc As tPdfDocument, _
    ByVal pageNo As Long, _
    ByVal fileNumber As Integer)

    Const METHOD_NAME As String = "WriteCrossingSection"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim dictCrossings As Object
    Dim dblLineX As Double
    Dim dblTextWidth As Double
    Dim lngIndex As Long
    Dim lngTextIndex As Long
    Dim strKey As String
    Dim varKey As Variant

    On Error GoTo ErrorHandler

    Set dictCrossings = CreateObject("Scripting.Dictionary")
    For lngIndex = 0 To pdfDoc.LineCount - 1
        If pdfDoc.Lines(lngIndex).PageNo = pageNo Then
            If Abs(pdfDoc.Lines(lngIndex).X1 - pdfDoc.Lines(lngIndex).X2) <= COORD_TOLERANCE Then
                dblLineX = (pdfDoc.Lines(lngIndex).X1 + pdfDoc.Lines(lngIndex).X2) / 2
                For lngTextIndex = 0 To pdfDoc.TextCount - 1
                    If pdfDoc.Texts(lngTextIndex).PageNo = pageNo Then
                        dblTextWidth = EstimateTextWidth(pdfDoc.Texts(lngTextIndex).TextValue, pdfDoc.Texts(lngTextIndex).FontName, pdfDoc.Texts(lngTextIndex).FontSize)
                        If dblLineX > pdfDoc.Texts(lngTextIndex).X + 0.5 And dblLineX < pdfDoc.Texts(lngTextIndex).X + dblTextWidth - 0.5 Then
                            If LineOverlapsTextVertically(pdfDoc.Lines(lngIndex), pdfDoc.Texts(lngTextIndex)) Then
                                strKey = "x=" & NumTxt(dblLineX) & "|lineY=" & NumTxt(MinDouble(pdfDoc.Lines(lngIndex).Y1, pdfDoc.Lines(lngIndex).Y2)) & ".." & NumTxt(MaxDouble(pdfDoc.Lines(lngIndex).Y1, pdfDoc.Lines(lngIndex).Y2)) & "|textX=" & NumTxt(pdfDoc.Texts(lngTextIndex).X) & "|textY=" & NumTxt(pdfDoc.Texts(lngTextIndex).Y) & "|font=" & pdfDoc.Texts(lngTextIndex).FontName & "@" & NumTxt(pdfDoc.Texts(lngTextIndex).FontSize) & "|text=" & SanitizeReportText(pdfDoc.Texts(lngTextIndex).TextValue)
                                If Not dictCrossings.Exists(strKey) Then
                                    dictCrossings.Add strKey, True
                                End If
                            End If
                        End If
                    End If
                Next lngTextIndex
            End If
        End If
    Next lngIndex

    Call EmitLine(fileNumber, "ESTIMATED_TEXT_CROSSINGS|count=" & CStr(dictCrossings.Count))
    For Each varKey In dictCrossings.Keys
        Call EmitLine(fileNumber, "CROSSING_TEXT|" & CStr(varKey))
    Next varKey

ExitSub:
    Set dictCrossings = Nothing
    Exit Sub

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "pageNo", _
        pageNo

    errorManager.save
    Resume ExitSub
End Sub

Private Sub WriteComparisonReport( _
    ByRef oldDoc As tPdfDocument, _
    ByRef newDoc As tPdfDocument, _
    ByVal outputPath As String)

    Const METHOD_NAME As String = "WriteComparisonReport"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim intFileNumber As Integer
    Dim lngCommonPages As Long
    Dim lngPage As Long

    On Error GoTo ErrorHandler

    intFileNumber = OpenNew(outputPath)
    Call EmitLine(intFileNumber, "PDF VECTOR COMPARISON V1")
    Call EmitLine(intFileNumber, "OLD|" & oldDoc.PdfPath)
    Call EmitLine(intFileNumber, "NEW|" & newDoc.PdfPath)
    Call EmitLine(intFileNumber, "OLD_DOCUMENT|version=" & oldDoc.PdfVersion & "|bytes=" & CStr(oldDoc.FileSize) & "|objects=" & CStr(oldDoc.ObjectCount) & "|xref=" & oldDoc.XrefKind & "|pages=" & CStr(oldDoc.PageCount))

    Call EmitLine(intFileNumber, "IMPORTANT|Use matching reports with the same transactions for meaningful text and page-level comparison.")

    lngCommonPages = oldDoc.PageCount
    If newDoc.PageCount < lngCommonPages Then
        lngCommonPages = newDoc.PageCount
    End If

    For lngPage = 1 To lngCommonPages
        Call ComparePage(oldDoc, newDoc, lngPage, intFileNumber)
    Next lngPage

    If oldDoc.PageCount <> newDoc.PageCount Then
        Call EmitLine(intFileNumber, "PAGE_COUNT_DIFFERENCE|old=" & CStr(oldDoc.PageCount) & "|new=" & CStr(newDoc.PageCount))
    End If

    End If
    Exit Sub

    Resume ExitSub
End Sub

    ByRef newDoc As tPdfDocument, _
    ByVal pageNo As Long, _
    ByVal fileNumber As Integer)

    Const METHOD_NAME As String = "ComparePage"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim newGeometry As Object
    Dim newStyles As Object
    Dim oldGeometry As Object
    Dim oldStyles As Object

    On Error GoTo ErrorHandler

    Call EmitLine(fileNumber, vbNullString)
    Call EmitLine(fileNumber, "PAGE_COMPARE|" & CStr(pageNo))
    Call EmitLine(fileNumber, "OLD_DICT|MediaBox=" & oldDoc.Pages(pageNo).MediaBox & "|CropBox=" & oldDoc.Pages(pageNo).CropBox & "|Rotate=" & oldDoc.Pages(pageNo).RotateValue & "|Fonts=" & oldDoc.Pages(pageNo).FontDictionary)

    Call EmitLine(fileNumber, "COUNTS|oldLines=" & CStr(oldDoc.Pages(pageNo).LineCount) & "|newLines=" & CStr(newDoc.Pages(pageNo).LineCount) & "|oldText=" & CStr(oldDoc.Pages(pageNo).TextCount) & "|newText=" & CStr(newDoc.Pages(pageNo).TextCount))

    Call WriteTargetComparison(oldDoc, newDoc, pageNo, TARGET_X_FIRST, fileNumber)
    Call WriteTargetComparison(oldDoc, newDoc, pageNo, TARGET_X_SECOND, fileNumber)

ExitSub:

    Set newGeometry = Nothing
    Set oldStyles = Nothing
    Set newStyles = Nothing
    Exit Sub

ErrorHandler:

    errDescription = VBA.Err.Description

        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _

        pageNo

    Resume ExitSub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    oldGeometry As Object, newGeometry As Object, oldStyles As Object, newStyles As Object, fileNumber As Integer
' Returns:       ---
' Description:   Writes missing, added, multiplicity, and style differences between line sets.
'-------------------------------------------------------------------------------
Private Sub WriteGeometryDifferences( _
    ByVal oldGeometry As Object, _
    ByVal newGeometry As Object, _
    ByVal oldStyles As Object, _
    ByVal newStyles As Object, _
    ByVal fileNumber As Integer)

    Const METHOD_NAME As String = "WriteGeometryDifferences"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngAdded As Long
    Dim lngChanged As Long
    Dim lngDetailCount As Long
    Dim lngMissing As Long
    Dim lngStyleChanged As Long
    Dim varKey As Variant

    On Error GoTo ErrorHandler

    For Each varKey In oldGeometry.Keys
        If Not newGeometry.Exists(varKey) Then
            lngMissing = lngMissing + 1
            If lngDetailCount < MAX_DETAIL_LINES Then
                lngDetailCount = lngDetailCount + 1
                Call EmitLine(fileNumber, "LINE_ONLY_OLD|count=" & CStr(oldGeometry(varKey)) & "|" & CStr(varKey))
            End If
        ElseIf CLng(oldGeometry(varKey)) <> CLng(newGeometry(varKey)) Then
            lngChanged = lngChanged + 1
            If lngDetailCount < MAX_DETAIL_LINES Then
                lngDetailCount = lngDetailCount + 1
                Call EmitLine(fileNumber, "LINE_MULTIPLICITY_CHANGED|old=" & CStr(oldGeometry(varKey)) & "|new=" & CStr(newGeometry(varKey)) & "|" & CStr(varKey))
            End If
        End If

        If newStyles.Exists(varKey) Then
            If CStr(oldStyles(varKey)) <> CStr(newStyles(varKey)) Then
                lngStyleChanged = lngStyleChanged + 1
                If lngDetailCount < MAX_DETAIL_LINES Then
                    lngDetailCount = lngDetailCount + 1
                    Call EmitLine(fileNumber, "LINE_STYLE_CHANGED|old=" & CStr(oldStyles(varKey)) & "|new=" & CStr(newStyles(varKey)) & "|" & CStr(varKey))
                End If
            End If
        End If
    Next varKey

    For Each varKey In newGeometry.Keys
        If Not oldGeometry.Exists(varKey) Then
            lngAdded = lngAdded + 1
            If lngDetailCount < MAX_DETAIL_LINES Then
                lngDetailCount = lngDetailCount + 1
                Call EmitLine(fileNumber, "LINE_ONLY_NEW|count=" & CStr(newGeometry(varKey)) & "|" & CStr(varKey))
            End If
        End If
    Next varKey

    Call EmitLine(fileNumber, "VECTOR_DIFF_SUMMARY|onlyOld=" & CStr(lngMissing) & "|onlyNew=" & CStr(lngAdded) & "|multiplicityChanged=" & CStr(lngChanged) & "|styleChanged=" & CStr(lngStyleChanged))

ExitSub:
    Exit Sub

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription

    errorManager.save
    Resume ExitSub
End Sub

' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    oldDoc As tPdfDocument, newDoc As tPdfDocument, pageNo As Long, targetX As Double, fileNumber As Integer
' Returns:       ---
' Description:   Compares stroke multiplicity, style, vertical span, and text crossings at one X coordinate.
'-------------------------------------------------------------------------------
Private Sub WriteTargetComparison( _
    ByRef oldDoc As tPdfDocument, _
    ByRef newDoc As tPdfDocument, _
    ByVal pageNo As Long, _
    ByVal targetX As Double, _
    ByVal fileNumber As Integer)

    Const METHOD_NAME As String = "WriteTargetComparison"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim lngNewCrossings As Long
    Dim lngNewOccurrences As Long
    Dim lngNewUnique As Long
    Dim lngOldCrossings As Long
    Dim lngOldOccurrences As Long
    Dim lngOldUnique As Long
    Dim strFinding As String
    Dim strNewDetails As String
    Dim strOldDetails As String

    On Error GoTo ErrorHandler

    Call TargetLineStats(oldDoc, pageNo, targetX, lngOldOccurrences, lngOldUnique, lngOldCrossings, strOldDetails)
    Call TargetLineStats(newDoc, pageNo, targetX, lngNewOccurrences, lngNewUnique, lngNewCrossings, strNewDetails)

    If lngNewOccurrences > lngOldOccurrences Then
        strFinding = strFinding & "NEW_HAS_MORE_STROKES;"
    End If
    If lngNewUnique > lngOldUnique Then
        strFinding = strFinding & "NEW_HAS_MORE_DISTINCT_SPANS_OR_STYLES;"
    End If
    If lngNewCrossings > lngOldCrossings Then
        strFinding = strFinding & "NEW_INTERSECTS_MORE_TEXT;"
    End If
    If strOldDetails <> strNewDetails Then
        strFinding = strFinding & "SPAN_OR_STYLE_DIFFERS;"
    End If
    If Len(strFinding) = 0 Then
        strFinding = "NO_TARGET_VECTOR_DIFFERENCE_DETECTED"
    End If

    Call EmitLine(fileNumber, "TARGET_COMPARE|x=" & NumTxt(targetX) & "|oldOccurrences=" & CStr(lngOldOccurrences) & "|newOccurrences=" & CStr(lngNewOccurrences) & "|oldUnique=" & CStr(lngOldUnique) & "|newUnique=" & CStr(lngNewUnique) & "|oldCrossings=" & CStr(lngOldCrossings) & "|newCrossings=" & CStr(lngNewCrossings) & "|finding=" & strFinding)
    Call EmitLine(fileNumber, "TARGET_OLD_DETAIL|x=" & NumTxt(targetX) & "|" & strOldDetails)
    Call EmitLine(fileNumber, "TARGET_NEW_DETAIL|x=" & NumTxt(targetX) & "|" & strNewDetails)

ExitSub:
    Exit Sub

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "pageNo;targetX", _
        pageNo, _
        targetX

    errorManager.save
    Resume ExitSub
End Sub

' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    pdfDoc As tPdfDocument, pageNo As Long, targetX As Double, occurrenceCount As Long, uniqueCount As Long, crossingCount As Long, details As String
' Returns:       ---
' Description:   Summarizes all stroked vertical rules at one target coordinate.
'-------------------------------------------------------------------------------
Private Sub TargetLineStats( _
    ByRef pdfDoc As tPdfDocument, _
    ByVal pageNo As Long, _
    ByVal targetX As Double, _
    ByRef occurrenceCount As Long, _
    ByRef uniqueCount As Long, _
    ByRef crossingCount As Long, _
    ByRef details As String)

    Const METHOD_NAME As String = "TargetLineStats"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim dictDetails As Object
    Dim lngIndex As Long
    Dim strKey As String
    Dim varKey As Variant

    On Error GoTo ErrorHandler

    Set dictDetails = CreateObject("Scripting.Dictionary")
    For lngIndex = 0 To pdfDoc.LineCount - 1
        If pdfDoc.Lines(lngIndex).PageNo = pageNo Then
            If IsVerticalAtX(pdfDoc.Lines(lngIndex), targetX) Then
                occurrenceCount = occurrenceCount + 1
                strKey = "y=" & NumTxt(MinDouble(pdfDoc.Lines(lngIndex).Y1, pdfDoc.Lines(lngIndex).Y2)) & ".." & NumTxt(MaxDouble(pdfDoc.Lines(lngIndex).Y1, pdfDoc.Lines(lngIndex).Y2)) & ",width=" & NumTxt(pdfDoc.Lines(lngIndex).LineWidth) & ",paint=" & pdfDoc.Lines(lngIndex).PaintOperator & ",source=" & pdfDoc.Lines(lngIndex).SourceOperator & ",color=" & pdfDoc.Lines(lngIndex).StrokeColor
                If dictDetails.Exists(strKey) Then
                    dictDetails(strKey) = CLng(dictDetails(strKey)) + 1
                Else
                    dictDetails.Add strKey, 1
                End If
            End If
        End If
    Next lngIndex

    uniqueCount = dictDetails.Count
    crossingCount = CountTargetTextCrossings(pdfDoc, pageNo, targetX, 0, False)
    For Each varKey In dictDetails.Keys
        If Len(details) > 0 Then
            details = details & "|"
        End If
        details = details & "count=" & CStr(dictDetails(varKey)) & "," & CStr(varKey)
    Next varKey
    If Len(details) = 0 Then
        details = "none"
    End If

ExitSub:
    Set dictDetails = Nothing
    Exit Sub

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "pageNo;targetX", _
        pageNo, _
        targetX

    errorManager.save
    Resume ExitSub
End Sub

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    pdfDoc As tPdfDocument, pageNo As Long, targetX As Double, fileNumber As Integer, writeDetails As Boolean
' Returns:       Long
' Description:   Counts estimated text bounding boxes intersected by a target vertical rule.
'-------------------------------------------------------------------------------
Private Function CountTargetTextCrossings( _
    ByRef pdfDoc As tPdfDocument, _
    ByVal pageNo As Long, _
    ByVal targetX As Double, _
    ByVal fileNumber As Integer, _
    ByVal writeDetails As Boolean) As Long

    Const METHOD_NAME As String = "CountTargetTextCrossings"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim dictCrossings As Object
    Dim dblTextWidth As Double
    Dim lngLineIndex As Long
    Dim lngTextIndex As Long
    Dim strKey As String
    Dim varKey As Variant

    On Error GoTo ErrorHandler

    Set dictCrossings = CreateObject("Scripting.Dictionary")
    For lngLineIndex = 0 To pdfDoc.LineCount - 1
        If pdfDoc.Lines(lngLineIndex).PageNo = pageNo Then
            If IsVerticalAtX(pdfDoc.Lines(lngLineIndex), targetX) Then
                For lngTextIndex = 0 To pdfDoc.TextCount - 1
                    If pdfDoc.Texts(lngTextIndex).PageNo = pageNo Then
                        dblTextWidth = EstimateTextWidth(pdfDoc.Texts(lngTextIndex).TextValue, pdfDoc.Texts(lngTextIndex).FontName, pdfDoc.Texts(lngTextIndex).FontSize)
                        If targetX > pdfDoc.Texts(lngTextIndex).X + 0.5 And targetX < pdfDoc.Texts(lngTextIndex).X + dblTextWidth - 0.5 Then
                            If LineOverlapsTextVertically(pdfDoc.Lines(lngLineIndex), pdfDoc.Texts(lngTextIndex)) Then
                                strKey = "textX=" & NumTxt(pdfDoc.Texts(lngTextIndex).X) & "|textY=" & NumTxt(pdfDoc.Texts(lngTextIndex).Y) & "|font=" & pdfDoc.Texts(lngTextIndex).FontName & "@" & NumTxt(pdfDoc.Texts(lngTextIndex).FontSize) & "|text=" & SanitizeReportText(pdfDoc.Texts(lngTextIndex).TextValue)
                                If Not dictCrossings.Exists(strKey) Then
                                    dictCrossings.Add strKey, True
                                End If
                            End If
                        End If
                    End If
                Next lngTextIndex
            End If
        End If
    Next lngLineIndex

    CountTargetTextCrossings = dictCrossings.Count
    If writeDetails And fileNumber > 0 Then
        For Each varKey In dictCrossings.Keys
            Call EmitLine(fileNumber, "TARGET_CROSSING_TEXT|x=" & NumTxt(targetX) & "|" & CStr(varKey))
        Next varKey
    End If

ExitFunction:
    Set dictCrossings = Nothing
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    CountTargetTextCrossings = 0

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "pageNo;targetX", _
        pageNo, _
        targetX

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka

' Parameters:    lineInfo As tPdfLine, textInfo As tPdfText
' Returns:       Boolean
' Description:   Returns whether a line vertically overlaps an estimated text box.
'-------------------------------------------------------------------------------
Private Function LineOverlapsTextVertically( _
    ByRef lineInfo As tPdfLine, _
    ByRef textInfo As tPdfText) As Boolean

    Const METHOD_NAME As String = "LineOverlapsTextVertically"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim dblLineBottom As Double
    Dim dblLineTop As Double
    Dim dblTextBottom As Double
    Dim dblTextTop As Double

    On Error GoTo ErrorHandler

    dblLineBottom = MinDouble(lineInfo.Y1, lineInfo.Y2)
    dblLineTop = MaxDouble(lineInfo.Y1, lineInfo.Y2)
    dblTextBottom = textInfo.Y - textInfo.FontSize * 0.25
    dblTextTop = textInfo.Y + textInfo.FontSize * 0.85
    LineOverlapsTextVertically = dblLineTop >= dblTextBottom And dblLineBottom <= dblTextTop

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    LineOverlapsTextVertically = False

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription

    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    textValue As String, fontName As String, fontSize As Double
' Returns:       Double
' Description:   Estimates a Times-family text run width for separator diagnostics.
'-------------------------------------------------------------------------------
Private Function EstimateTextWidth( _
    ByVal textValue As String, _
    ByVal fontName As String, _
    ByVal fontSize As Double) As Double

    Const METHOD_NAME As String = "EstimateTextWidth"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim dblFactor As Double
    Dim dblUnits As Double
    Dim lngIndex As Long
    Dim strCharacter As String

    On Error GoTo ErrorHandler

    For lngIndex = 1 To Len(textValue)
        strCharacter = Mid$(textValue, lngIndex, 1)
        Select Case strCharacter
            Case " "
                dblUnits = dblUnits + 0.25
            Case "i", "l", "I", ".", ",", ":", ";", "!", "'", "|"
                dblUnits = dblUnits + 0.25
            Case "M", "W", "m", "w", "@", "%"
                dblUnits = dblUnits + 0.85
            Case "0" To "9"
                dblUnits = dblUnits + 0.5
            Case "A" To "Z"
                dblUnits = dblUnits + 0.65
            Case Else
                dblUnits = dblUnits + 0.48
        End Select
    Next lngIndex

    dblFactor = 1
    If InStr(1, fontName, "Bold", vbTextCompare) > 0 Then
        dblFactor = 1.03
    End If
    EstimateTextWidth = dblUnits * fontSize * dblFactor

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    EstimateTextWidth = Len(textValue) * fontSize * 0.5

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "fontName;fontSize", _
        fontName, _
        fontSize

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    lineInfo As tPdfLine
' Returns:       String
' Description:   Returns a normalized coordinate-only key for one horizontal or vertical line.
'-------------------------------------------------------------------------------
Private Function GeometryKey(ByRef lineInfo As tPdfLine) As String
    Const METHOD_NAME As String = "GeometryKey"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim dblFirst As Double
    Dim dblSecond As Double

    On Error GoTo ErrorHandler

    If Abs(lineInfo.X1 - lineInfo.X2) <= COORD_TOLERANCE Then
        dblFirst = MinDouble(lineInfo.Y1, lineInfo.Y2)
        dblSecond = MaxDouble(lineInfo.Y1, lineInfo.Y2)
        GeometryKey = "V|x=" & NumTxt((lineInfo.X1 + lineInfo.X2) / 2) & "|y=" & NumTxt(dblFirst) & ".." & NumTxt(dblSecond)
    Else
        dblFirst = MinDouble(lineInfo.X1, lineInfo.X2)
        dblSecond = MaxDouble(lineInfo.X1, lineInfo.X2)
        GeometryKey = "H|y=" & NumTxt((lineInfo.Y1 + lineInfo.Y2) / 2) & "|x=" & NumTxt(dblFirst) & ".." & NumTxt(dblSecond)
    End If

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    GeometryKey = "ERROR"

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    data As Byte array, fromIndex As Long, toIndex As Long
' Returns:       String
' Description:   Converts a byte range to a VBA string.
'-------------------------------------------------------------------------------
Private Function BytesText( _
    ByRef data() As Byte, _
    ByVal fromIndex As Long, _
    ByVal toIndex As Long) As String

    Const METHOD_NAME As String = "BytesText"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim arrTemporary() As Byte
    Dim lngIndex As Long
    Dim lngLength As Long

    On Error GoTo ErrorHandler

    If fromIndex < 0 Then
        fromIndex = 0
    End If
    If toIndex > UBound(data) Then
        toIndex = UBound(data)
    End If
    If toIndex < fromIndex Then
        GoTo ExitFunction
    End If

    lngLength = toIndex - fromIndex + 1
    ReDim arrTemporary(0 To lngLength - 1)
    For lngIndex = 0 To lngLength - 1
        arrTemporary(lngIndex) = data(fromIndex + lngIndex)
    Next lngIndex
    BytesText = StrConv(arrTemporary, vbUnicode)

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    BytesText = vbNullString

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "fromIndex;toIndex", _
        fromIndex, _
        toIndex

    errorManager.save
    Resume ExitFunction
End Function

'-------------------------------------------------------------------------------
' Author:        Pawel Ligezka
' Creation date: 2026-08-06
' Parameters:    data As Byte array, startAt As Long, pattern As String
' Returns:       Long
' Description:   Finds an ASCII byte pattern in a PDF byte array.
'-------------------------------------------------------------------------------
Private Function FindBytes( _
    ByRef data() As Byte, _
    ByVal startAt As Long, _
    ByVal pattern As String) As Long

    Const METHOD_NAME As String = "FindBytes"
    Dim errorManager As New ErrorManager
    Dim errDescription As String
    Dim errNumber As Long
    Dim arrPattern() As Byte
    Dim lngDataIndex As Long
    Dim lngPatternIndex As Long
    Dim lngPatternLength As Long

    On Error GoTo ErrorHandler

    arrPattern = StrConv(pattern, vbFromUnicode)
    lngPatternLength = UBound(arrPattern) - LBound(arrPattern) + 1
    If startAt < 0 Then
        startAt = 0
    End If

    For lngDataIndex = startAt To UBound(data) - lngPatternLength + 1
        If data(lngDataIndex) = arrPattern(0) Then
            For lngPatternIndex = 1 To lngPatternLength - 1
                If data(lngDataIndex + lngPatternIndex) <> arrPattern(lngPatternIndex) Then
                    Exit For
                End If
            Next lngPatternIndex
            If lngPatternIndex = lngPatternLength Then
                FindBytes = lngDataIndex
                GoTo ExitFunction
            End If
        End If
    Next lngDataIndex

    FindBytes = -1

ExitFunction:
    Exit Function

ErrorHandler:
    errNumber = VBA.Err.Number
    errDescription = VBA.Err.Description
    FindBytes = -1

    errorManager.addError _
        CLASS_NAME, _
        METHOD_NAME, _
        errNumber, _
        errDescription, _
        "startAt;pattern", _
        startAt, _
        pattern

    errorManager.save
    Resume ExitFunction
End Function
